cmake_minimum_required(VERSION 3.22)
project(rtos_stdcpp_cortex_m LANGUAGES C CXX ASM)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-ffunction-sections -fdata-sections)
set(RTOS_STDCPP_CXX_STANDARD 17 CACHE STRING "Language mode")
set(CMAKE_CXX_STANDARD ${RTOS_STDCPP_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(THREADX_KERNEL_ROOT "" CACHE PATH "Pinned official ThreadX kernel")
set(RTOS_STDCPP_CORTEX_M "m4" CACHE STRING "m4, m7 or m55")
option(RTOS_STDCPP_M_TICK_WRAP "Seed kernel tick near wrap in qualification" OFF)
get_filename_component(root "../.." ABSOLUTE)
set(RTOS_STDCPP_BACKEND "ThreadX" CACHE STRING "ThreadX or FreeRTOS")
set(FREERTOS_KERNEL_ROOT "" CACHE PATH "Pinned official FreeRTOS kernel")
if(RTOS_STDCPP_BACKEND STREQUAL "ThreadX")
    if(NOT EXISTS "${THREADX_KERNEL_ROOT}/common/inc/tx_api.h")
        message(FATAL_ERROR "Supply THREADX_KERNEL_ROOT; builds never download dependencies")
    endif()
    set(port "${THREADX_KERNEL_ROOT}/ports/cortex_${RTOS_STDCPP_CORTEX_M}/gnu")
    file(GLOB common_sources CONFIGURE_DEPENDS "${THREADX_KERNEL_ROOT}/common/src/*.c")
    file(GLOB port_sources CONFIGURE_DEPENDS "${port}/src/*.S" "${port}/src/*.c")
    list(FILTER port_sources EXCLUDE REGEX "/tx_initialize_low_level\\.S$")
    add_library(kernel OBJECT ${common_sources} ${port_sources})
    target_include_directories(kernel PUBLIC "${THREADX_KERNEL_ROOT}/common/inc" "${port}/inc")
    target_compile_definitions(kernel PUBLIC TX_INCLUDE_USER_DEFINE_FILE)
    set(RTOS_STDCPP_THREADX_TARGET kernel)
    set(backend_platform threadx.cpp)
else()
    if(NOT RTOS_STDCPP_BACKEND STREQUAL "FreeRTOS" OR NOT EXISTS "${FREERTOS_KERNEL_ROOT}/include/FreeRTOS.h")
        message(FATAL_ERROR "Select ThreadX or supply FREERTOS_KERNEL_ROOT for FreeRTOS")
    endif()
    set(port_name ARM_CM4F)
    if(RTOS_STDCPP_CORTEX_M STREQUAL "m7")
        set(port_name ARM_CM7/r0p1)
    elseif(RTOS_STDCPP_CORTEX_M STREQUAL "m55")
        set(port_name ARM_CM55_NTZ/non_secure)
    endif()
    set(port "${FREERTOS_KERNEL_ROOT}/portable/GCC/${port_name}")
    set(port_sources "${port}/port.c")
    if(RTOS_STDCPP_CORTEX_M STREQUAL "m55")
        list(APPEND port_sources "${port}/portasm.c")
    endif()
    add_library(kernel OBJECT "${FREERTOS_KERNEL_ROOT}/tasks.c" "${FREERTOS_KERNEL_ROOT}/queue.c"
        "${FREERTOS_KERNEL_ROOT}/list.c" "${FREERTOS_KERNEL_ROOT}/event_groups.c"
        "${FREERTOS_KERNEL_ROOT}/portable/MemMang/heap_4.c" ${port_sources})
    target_include_directories(kernel PUBLIC "${FREERTOS_KERNEL_ROOT}/include" "${port}")
    set(RTOS_STDCPP_FREERTOS_TARGET kernel)
    set(backend_platform freertos.cpp)
    if(RTOS_STDCPP_M_TICK_WRAP)
        target_compile_definitions(kernel PUBLIC configINITIAL_TICK_COUNT=0xffffffe2U)
    endif()
endif()
target_include_directories(kernel PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set(RTOS_STDCPP_PROFILE qemu-${RTOS_STDCPP_CORTEX_M}-gcc-14.3.1-cxx${RTOS_STDCPP_CXX_STANDARD} CACHE STRING "" FORCE)
set(RTOS_STDCPP_BUILD_TEST_SUPPORT ON)
add_subdirectory("${root}" library)
target_link_libraries(kernel PRIVATE rtos-stdcpp::config)
include("${root}/cmake/runtime.cmake")
rtos_stdcpp_add_runtime(runtime BACKEND ${RTOS_STDCPP_BACKEND} CONFIG_TARGET kernel)
target_link_libraries(runtime PRIVATE rtos-stdcpp::config)
set_target_properties(runtime PROPERTIES CXX_STANDARD ${RTOS_STDCPP_CXX_STANDARD})
set(ram_address 0x20000000)
set(cpu_hz 25000000)
if(RTOS_STDCPP_CORTEX_M STREQUAL "m55")
    set(ram_address 0x21000000)
    set(cpu_hz 32000000)
endif()
configure_file(linker.ld.in linker.ld @ONLY)
add_library(platform OBJECT startup.S platform.cpp ${backend_platform} "${root}/tests/qualification/backend.cpp")
target_compile_definitions(kernel PUBLIC RTOS_STDCPP_CPU_HZ=${cpu_hz})
target_compile_definitions(platform PUBLIC RTOS_STDCPP_TLS_CUSTOM_ALLOCATOR=$<BOOL:${RTOS_STDCPP_TLS_CUSTOM_ALLOCATOR}>)
target_include_directories(platform PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${root}/tests/qualification")
target_link_libraries(platform PUBLIC kernel rtos-stdcpp::config PRIVATE rtos-stdcpp::test-support)
set_target_properties(platform PROPERTIES CXX_STANDARD ${RTOS_STDCPP_CXX_STANDARD})
set(runtime_objects runtime)
set(linker_script "${CMAKE_CURRENT_BINARY_DIR}/linker.ld")
set(fpu_probe "${root}/tests/qualification/fpu_m.S")
set(qemu_machine ${RTOS_STDCPP_CORTEX_M})
set(test_tick_wrap ${RTOS_STDCPP_M_TICK_WRAP})
enable_testing()
include("${root}/cmake/qemu-tests.cmake")
rtos_stdcpp_qemu_images()
