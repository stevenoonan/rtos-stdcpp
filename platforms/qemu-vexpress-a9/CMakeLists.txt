cmake_minimum_required(VERSION 3.22)
project(rtos_stdcpp_a9 LANGUAGES C CXX ASM)
set(RTOS_STDCPP_CXX_STANDARD 17 CACHE STRING "C++ language mode: 17 or 20")
set(CMAKE_CXX_STANDARD ${RTOS_STDCPP_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-ffunction-sections -fdata-sections)
option(RTOS_STDCPP_A9_TICK_WRAP "Start the test scheduler 30 ticks before wrap" OFF)
if(RTOS_STDCPP_A9_TICK_WRAP)
    add_compile_definitions(configINITIAL_TICK_COUNT=0xffffffe2U RTOS_STDCPP_A9_TICK_WRAP=1)
endif()
get_filename_component(root "../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(kernel "${root}/third_party/freertos-kernel")
set(port "${kernel}/portable/GCC/ARM_CA9")
add_library(freertos OBJECT "${kernel}/tasks.c" "${kernel}/queue.c" "${kernel}/list.c" "${kernel}/event_groups.c"
    "${kernel}/portable/MemMang/heap_4.c" "${port}/port.c" "${port}/portASM.S")
# The pinned 10.4.3 port omits string.h in its full-FPU path.
set_source_files_properties("${port}/port.c" PROPERTIES COMPILE_OPTIONS "-include;string.h")
target_include_directories(freertos PUBLIC "${kernel}/include" "${port}" "${CMAKE_CURRENT_SOURCE_DIR}")
set(RTOS_STDCPP_PROFILE qemu-a9-gcc-14.3.1-cxx${RTOS_STDCPP_CXX_STANDARD} CACHE STRING "" FORCE)
set(RTOS_STDCPP_FREERTOS_TARGET freertos)
set(RTOS_STDCPP_BUILD_TEST_SUPPORT ON)
add_subdirectory("${root}" library)
target_link_libraries(freertos PRIVATE rtos-stdcpp::config)
add_library(platform OBJECT startup_ARMCA9.cpp system_ARMCA9.c cmsis/irq_ctrl_gic.c
    ca9_global_timer.c FreeRTOS_ca9_hooks.c console.cpp runtime.cpp cxx_guards.cpp cxx_nothrow.cpp test_platform.cpp)
target_include_directories(platform PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/cmsis")
target_link_libraries(platform PUBLIC freertos rtos-stdcpp::config PRIVATE rtos-stdcpp::test-support)
if(RTOS_STDCPP_TLS_CUSTOM_ALLOCATOR)
    target_sources(platform PRIVATE tls_allocator.cpp)
    target_compile_definitions(platform PUBLIC RTOS_STDCPP_TEST_CUSTOM_TLS_ALLOCATOR=1)
endif()

function(image name)
    add_executable(${name} ${ARGN})
    set_target_properties(${name} PROPERTIES SUFFIX .elf)
    target_link_libraries(${name} PRIVATE platform freertos rtos-stdcpp::rtos-stdcpp rtos-stdcpp::test-support)
    target_link_options(${name} PRIVATE -marm -mcpu=cortex-a9 -mfpu=vfpv3 -mfloat-abi=hard
        --specs=nosys.specs -nostartfiles "-T${CMAKE_CURRENT_SOURCE_DIR}/linker.ld"
        "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/${name}.map,--cref,--gc-sections")
    set_property(TARGET ${name} APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/linker.ld")
endfunction()
image(conformance "${root}/tests/support/embedded_main.cpp")
if(RTOS_STDCPP_CXX_STANDARD EQUAL 20)
    image(cxx20 "${root}/tests/support/embedded_main.cpp")
    target_compile_definitions(cxx20 PRIVATE RTOS_STDCPP_RUN_CXX20=1)
endif()
image(qualification qualification.cpp)
image(regression regression_main.cpp "${root}/tests/regression/lifecycle_regression.cpp")
target_include_directories(regression PRIVATE "${root}/tests/regression")
foreach(mode THROWING_NEW KERNEL_ALLOC JOINABLE RECURSIVE_GUARD TLS_STARTUP TLS_NATIVE TLS_ALLOC TLS_OVERFLOW)
    string(TOLOWER "${mode}" name)
    image(fail_${name} failure.cpp)
    target_compile_definitions(fail_${name} PRIVATE FAIL_${mode}=1 FAILURE_NAME="${name}")
endforeach()
enable_testing()
set(RTOS_STDCPP_QEMU "" CACHE FILEPATH "Checksum-qualified qemu-system-arm executable")
if(RTOS_STDCPP_QEMU)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    foreach(pair "conformance;cxx17" "qualification;a9" "regression;regression")
        list(GET pair 0 executable)
        list(GET pair 1 suite)
        add_test(NAME ${executable} COMMAND "${Python3_EXECUTABLE}" "${root}/tools/run_qemu.py"
            --qemu "${RTOS_STDCPP_QEMU}" --image $<TARGET_FILE:${executable}> --suite ${suite}
            --output "${CMAKE_CURRENT_BINARY_DIR}/${executable}.json")
    endforeach()
    if(RTOS_STDCPP_CXX_STANDARD EQUAL 20)
        add_test(NAME cxx20 COMMAND "${Python3_EXECUTABLE}" "${root}/tools/run_qemu.py"
            --qemu "${RTOS_STDCPP_QEMU}" --image $<TARGET_FILE:cxx20> --suite cxx20
            --output "${CMAKE_CURRENT_BINARY_DIR}/cxx20.json")
        set_tests_properties(cxx20 PROPERTIES LABELS emulator TIMEOUT 35)
    endif()
    foreach(mode throwing_new kernel_alloc joinable recursive_guard tls_startup tls_native tls_alloc tls_overflow)
        set(allocator_arguments)
        if(RTOS_STDCPP_TLS_CUSTOM_ALLOCATOR AND mode STREQUAL "tls_alloc")
            list(APPEND allocator_arguments --custom-tls-allocator)
        endif()
        add_test(NAME fail_${mode} COMMAND "${Python3_EXECUTABLE}" "${root}/tools/run_qemu.py"
            --qemu "${RTOS_STDCPP_QEMU}" --image $<TARGET_FILE:fail_${mode}> --failure ${mode}
            ${allocator_arguments}
            --output "${CMAKE_CURRENT_BINARY_DIR}/fail_${mode}.json")
    endforeach()
    add_test(NAME link_provenance COMMAND "${Python3_EXECUTABLE}" "${root}/tools/check_a9.py"
        --build "${CMAKE_CURRENT_BINARY_DIR}" --toolchain "${RTOS_STDCPP_TOOLCHAIN_ROOT}")
    set_tests_properties(conformance qualification regression fail_throwing_new fail_kernel_alloc fail_joinable fail_recursive_guard
        fail_tls_startup fail_tls_native fail_tls_alloc fail_tls_overflow
        PROPERTIES LABELS emulator TIMEOUT 35)
    set_tests_properties(link_provenance PROPERTIES LABELS compile_link TIMEOUT 30)
endif()
