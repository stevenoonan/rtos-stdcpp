cmake_minimum_required(VERSION 3.22)
project(rtos_stdcpp_a9 LANGUAGES C CXX ASM)
set(RTOS_STDCPP_CXX_STANDARD 17 CACHE STRING "C++ language mode: 17 or 20")
set(CMAKE_CXX_STANDARD ${RTOS_STDCPP_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-ffunction-sections -fdata-sections)
option(RTOS_STDCPP_A9_TICK_WRAP "Start the test scheduler 30 ticks before wrap" OFF)
if(RTOS_STDCPP_A9_TICK_WRAP)
    add_compile_definitions(configINITIAL_TICK_COUNT=0xffffffe2U RTOS_STDCPP_A9_TICK_WRAP=1)
endif()
get_filename_component(root "../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(kernel "${root}/third_party/freertos-kernel")
set(port "${kernel}/portable/GCC/ARM_CA9")
add_library(kernel OBJECT "${kernel}/tasks.c" "${kernel}/queue.c" "${kernel}/list.c" "${kernel}/event_groups.c"
    "${kernel}/portable/MemMang/heap_4.c" "${port}/port.c" "${port}/portASM.S")
# The pinned 10.4.3 port omits string.h in its full-FPU path.
set_source_files_properties("${port}/port.c" PROPERTIES COMPILE_OPTIONS "-include;string.h")
target_include_directories(kernel PUBLIC "${kernel}/include" "${port}" "${CMAKE_CURRENT_SOURCE_DIR}")
set(RTOS_STDCPP_PROFILE qemu-a9-gcc-14.3.1-cxx${RTOS_STDCPP_CXX_STANDARD} CACHE STRING "" FORCE)
set(RTOS_STDCPP_BACKEND FreeRTOS)
set(RTOS_STDCPP_FREERTOS_TARGET kernel)
set(RTOS_STDCPP_BUILD_TEST_SUPPORT ON)
add_subdirectory("${root}" library)
target_link_libraries(kernel PRIVATE rtos-stdcpp::config)
add_library(platform OBJECT startup_ARMCA9.cpp system_ARMCA9.c cmsis/irq_ctrl_gic.c
    ca9_global_timer.c FreeRTOS_ca9_hooks.c console.cpp runtime.cpp cxx_guards.cpp cxx_nothrow.cpp test_platform.cpp "${root}/tests/qualification/backend.cpp")
target_include_directories(platform PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/cmsis")
target_link_libraries(platform PUBLIC kernel rtos-stdcpp::config PRIVATE rtos-stdcpp::test-support)
set(linker_script "${CMAKE_CURRENT_SOURCE_DIR}/linker.ld")
set(fpu_probe "${root}/tests/qualification/fpu_a9.S")
set(qemu_machine a9)
set(test_tick_wrap ${RTOS_STDCPP_A9_TICK_WRAP})
enable_testing()
include("${root}/cmake/qemu-tests.cmake")
rtos_stdcpp_qemu_images()
