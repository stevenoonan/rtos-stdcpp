find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/catalog.json" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/generate_test_catalog.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/generate_test_catalog.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/catalog.json" "${generated}/test_catalog.h"
    RESULT_VARIABLE generated_result)
if(NOT generated_result EQUAL 0)
    message(FATAL_ERROR "Test catalog generation failed")
endif()
add_library(rtos_stdcpp_test_support OBJECT
    support/runner.cpp conformance/standard.cpp conformance/facilities.cpp conformance/extended.cpp)
add_library(rtos-stdcpp::test-support ALIAS rtos_stdcpp_test_support)
target_include_directories(rtos_stdcpp_test_support PUBLIC support "${generated}")
target_compile_features(rtos_stdcpp_test_support PUBLIC cxx_std_${RTOS_STDCPP_CXX_STANDARD})
set_target_properties(rtos_stdcpp_test_support PROPERTIES CXX_STANDARD ${RTOS_STDCPP_CXX_STANDARD} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(RTOS_STDCPP_CXX_STANDARD EQUAL 20)
    target_sources(rtos_stdcpp_test_support PRIVATE conformance/cxx20.cpp)
endif()
if(RTOS_STDCPP_BUILD_LIBRARY)
    target_compile_definitions(rtos_stdcpp_test_support PUBLIC RTOS_STDCPP_TEST_FREERTOS=1)
    target_link_libraries(rtos_stdcpp_test_support PUBLIC rtos-stdcpp::config
        PRIVATE "${RTOS_STDCPP_FREERTOS_TARGET}")
endif()
if(RTOS_STDCPP_BUILD_HOST_TESTS)
    if(CMAKE_CROSSCOMPILING)
        message(FATAL_ERROR "Host tests require a native compiler; embedded consumers select test support separately")
    endif()
    find_package(Threads REQUIRED)
    target_link_libraries(rtos_stdcpp_test_support PUBLIC Threads::Threads)
    target_compile_options(rtos_stdcpp_test_support PRIVATE -Wall -Wextra -Werror -fsanitize=undefined -fno-sanitize-recover=all)
    target_link_options(rtos_stdcpp_test_support PUBLIC -fsanitize=undefined)
    add_library(test_host_platform OBJECT host/platform.cpp)
    target_link_libraries(test_host_platform PRIVATE rtos_stdcpp_test_support)
    add_executable(cxx17_reference host/main.cpp)
    target_link_libraries(cxx17_reference PRIVATE rtos_stdcpp_test_support test_host_platform)
    if(RTOS_STDCPP_CXX_STANDARD EQUAL 20)
        add_executable(cxx20_reference host/main.cpp)
        target_compile_definitions(cxx20_reference PRIVATE RTOS_STDCPP_RUN_CXX20=1)
        target_link_libraries(cxx20_reference PRIVATE rtos_stdcpp_test_support test_host_platform)
        add_test(NAME cxx20_reference COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/run_tests.py"
            --suite cxx20 --layer host --timeout 30 -- $<TARGET_FILE:cxx20_reference>)
        set_tests_properties(cxx20_reference PROPERTIES LABELS "host;reference" TIMEOUT 35)
    endif()
    add_executable(runner_fixture support/fixture.cpp)
    target_link_libraries(runner_fixture PRIVATE rtos_stdcpp_test_support test_host_platform)
    foreach(target cxx17_reference runner_fixture test_host_platform)
        set_target_properties(${target} PROPERTIES CXX_STANDARD ${RTOS_STDCPP_CXX_STANDARD} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    endforeach()
    add_test(NAME cxx17_reference COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/run_tests.py"
        --suite cxx17 --layer host --timeout 20 -- $<TARGET_FILE:cxx17_reference>)
    set_tests_properties(cxx17_reference PROPERTIES LABELS "host;reference" TIMEOUT 25)
    add_test(NAME runner_negative_controls COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/support/check_runner.py"
        --fixture $<TARGET_FILE:runner_fixture>)
    add_test(NAME runner_negative_controls_optimized COMMAND "${Python3_EXECUTABLE}" -O "${CMAKE_CURRENT_SOURCE_DIR}/support/check_runner.py"
        --fixture $<TARGET_FILE:runner_fixture>)
    set_tests_properties(runner_negative_controls runner_negative_controls_optimized PROPERTIES LABELS "host;harness" TIMEOUT 20)
    add_subdirectory(unit)
endif()
